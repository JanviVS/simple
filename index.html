React Native BMI Calculator Code

import React, { useState } from 'react'; import { View, Text, TextInput, Button, StyleSheet } from 'react-native'; export default function App() { const [weight, setWeight] = useState(''); const [height, setHeight] = useState(''); const [bmi, setBmi] = useState(''); const calculateBmi = () => { const w = parseFloat(weight), h = parseFloat(height) / 100; setBmi(!w || !h ? 'Error: Enter valid numbers' : `BMI: ${(w / (h * h)).toFixed(2)}`); }; return ( <View style={styles.container}> <Text style={styles.title}>BMI Calculator</Text> <TextInput style={styles.input} placeholder="Weight (kg)" keyboardType="numeric" onChangeText={setWeight} /> <TextInput style={styles.input} placeholder="Height (cm)" keyboardType="numeric" onChangeText={setHeight} /> <Button title="Calculate" onPress={calculateBmi} /> <Text style={styles.result}>{bmi}</Text> </View> ); } const styles = StyleSheet.create({ container: { padding: 10, marginTop: 50, }, title: { fontSize: 20, marginBottom: 20, }, input: { borderWidth: 1, marginBottom: 10, padding: 8, }, result: { fontSize: 18, marginTop: 20, }, });
